//@version=6
strategy("Стратегия Dynamic Levels с Кластерным Анализом", overlay=true, default_qty_type=strategy.percent_of_equity, default_qty_value=2, initial_capital=10000)

// Параметры для Dynamic Levels Channel
dlc_period = input.int(40, title="Период для Dynamic Levels Channel", minval=10, maxval=200)
poc_lookback = input.int(5, title="Период для определения направления POC", minval=3, maxval=20)
value_area_percent = input.float(70.0, title="Процент объема в Value Area", minval=50.0, maxval=90.0, step=1.0)

// Параметры для Nadaraya-Watson Envelope
h = input.float(8., title="NWE - Bandwidth", minval=0, group="Nadaraya-Watson Envelope")
mult = input.float(3., title="NWE - Множитель", minval=0, group="Nadaraya-Watson Envelope")
nwe_src = input(close, title="NWE - Источник", group="Nadaraya-Watson Envelope")
use_nwe = input.bool(true, title="Использовать Nadaraya-Watson Envelope", group="Nadaraya-Watson Envelope")
repaint = input.bool(false, title="NWE - Перерисовка", tooltip="Отключение перерисовки сделает индикатор более стабильным", group="Nadaraya-Watson Envelope")

// Параметры для Cluster Search
min_volume_threshold = input.float(1.5, title="Минимальный порог объема (множитель от среднего)", minval=1.0, maxval=5.0, step=0.1)
delta_threshold = input.float(0.7, title="Порог дельты для кластеров", minval=0.5, maxval=0.9, step=0.05)

// Параметры управления капиталом
risk_percent = input.float(2.0, title="Риск на сделку (%)", minval=0.1, maxval=5.0, step=0.1)
stop_loss_atr = input.float(2.0, title="Stop Loss (ATR множитель)", minval=0.5, maxval=5.0, step=0.1)
take_profit_atr = input.float(5.0, title="Take Profit (ATR множитель)", minval=1.0, maxval=10.0, step=0.1)
atr_period = input.int(14, title="ATR период", minval=5, maxval=50)
max_trades_per_day = input.int(2, title="Максимум сделок в день", minval=1, maxval=10)
use_trailing_stop = input.bool(true, title="Использовать трейлинг-стоп")
trailing_step = input.float(1.0, title="Шаг трейлинг-стопа (ATR множитель)", minval=0.1, maxval=5.0, step=0.1) // увеличено по умолчанию
trail_offset_mult = input.float(2.0, title="Множитель трейлинг-стопа (ATR)", minval=1.0, maxval=5.0, step=0.1) // новый параметр

// Расчет ATR
atr_value = ta.atr(atr_period)

// Расчет объемного профиля для имитации Dynamic Levels Channel
// Используем Volume Profile для определения POC, VAH и VAL
var float[] volume_profile = array.new_float(100, 0.0)
var float poc_price = na
var float vah_price = na
var float val_price = na
var float prev_poc_price = na
var float[] poc_history = array.new_float(poc_lookback, 0.0)

// Предварительный расчет значений для использования в функции
lowest_low_dlc = ta.lowest(low, dlc_period)
highest_high_dlc = ta.highest(high, dlc_period)

// Функция для обновления объемного профиля
update_volume_profile() =>
    float local_poc_price = na
    float local_vah_price = na
    float local_val_price = na
    float local_prev_poc_price = prev_poc_price
    
    price_range = highest_high_dlc - lowest_low_dlc
    if price_range <= 0
        price_range := high - low
    
    price_step = price_range / 100
    
    // Очистка массива
    array.fill(volume_profile, 0.0)
    
    // Заполнение профиля объемами
    for i = 0 to dlc_period - 1
        bar_high = high[i]
        bar_low = low[i]
        bar_volume = volume[i]
        
        bar_range = bar_high - bar_low
        if bar_range > 0
            volume_per_tick = bar_volume / bar_range
            
            start_idx = math.floor((bar_low - lowest_low_dlc) / price_step)
            end_idx = math.ceil((bar_high - lowest_low_dlc) / price_step)
            
            start_idx := math.max(0, math.min(99, start_idx))
            end_idx := math.max(0, math.min(99, end_idx))
            
            for j = start_idx to end_idx
                array.set(volume_profile, j, array.get(volume_profile, j) + volume_per_tick)
    
    // Нахождение POC (Point of Control)
    max_volume = 0.0
    poc_idx = 0
    
    for i = 0 to 99
        vol = array.get(volume_profile, i)
        if vol > max_volume
            max_volume := vol
            poc_idx := i
    
    // Расчет POC цены
    local_prev_poc_price := poc_price
    local_poc_price := lowest_low_dlc + poc_idx * price_step
    
    // Расчет Value Area (VAH и VAL)
    total_volume = array.sum(volume_profile)
    target_volume = total_volume * value_area_percent / 100
    
    current_volume = array.get(volume_profile, poc_idx)
    upper_idx = poc_idx
    lower_idx = poc_idx
    
    while current_volume < target_volume and (upper_idx < 99 or lower_idx > 0)
        upper_vol = upper_idx < 99 ? array.get(volume_profile, upper_idx + 1) : 0
        lower_vol = lower_idx > 0 ? array.get(volume_profile, lower_idx - 1) : 0
        
        if upper_vol >= lower_vol and upper_idx < 99
            upper_idx += 1
            current_volume += upper_vol
        else if lower_idx > 0
            lower_idx -= 1
            current_volume += lower_vol
    
    local_vah_price := lowest_low_dlc + upper_idx * price_step
    local_val_price := lowest_low_dlc + lower_idx * price_step
    
    [local_poc_price, local_vah_price, local_val_price, local_prev_poc_price]

// Обновление профиля объема на каждом баре
[updated_poc_price, updated_vah_price, updated_val_price, updated_prev_poc_price] = update_volume_profile()
prev_poc_price := updated_prev_poc_price
poc_price := updated_poc_price
vah_price := updated_vah_price
val_price := updated_val_price

// Добавление POC в историю
array.push(poc_history, poc_price)
if array.size(poc_history) > poc_lookback
    array.shift(poc_history)

// Определение направления тренда по POC
poc_direction = 0
if array.size(poc_history) > 0
    if array.get(poc_history, array.size(poc_history) - 1) > array.get(poc_history, 0)
        poc_direction := 1
    else if array.get(poc_history, array.size(poc_history) - 1) < array.get(poc_history, 0)
        poc_direction := -1

// Расчет средних объемов для определения кластеров
avg_volume = ta.sma(volume, 20)
high_volume = volume > avg_volume * min_volume_threshold

// Расчет дельты (разница между покупками и продажами)
// В Pine Script нет прямого доступа к данным о покупках/продажах, поэтому используем приближение
delta = 0.0
if close > open
    delta := volume * (close - low) / (high - low) - volume * (high - close) / (high - low)
else
    delta := volume * (close - low) / (high - low) - volume * (high - open) / (high - low)

delta_ratio = math.abs(delta) / volume
strong_delta = delta_ratio > delta_threshold

// Функции для Nadaraya-Watson Envelope
// Gaussian window
gauss(x, h) => math.exp(-(math.pow(x, 2)/(h * h * 2)))

// Расчет Nadaraya-Watson Envelope
var nwe_coefs = array.new_float(0)
var nwe_den = 0.

if barstate.isfirst and not repaint
    for i = 0 to 499
        w = gauss(i, h)
        array.push(nwe_coefs, w)
    nwe_den := array.sum(nwe_coefs)

nwe_out = 0.
if not repaint
    for i = 0 to 499
        nwe_out += nwe_src[i] * array.get(nwe_coefs, i)
    nwe_out /= nwe_den
    
nwe_mae = ta.sma(math.abs(nwe_src - nwe_out), 499) * mult

nwe_upper = nwe_out + nwe_mae
nwe_lower = nwe_out - nwe_mae

// Сигналы от Nadaraya-Watson Envelope
nwe_buy_signal = ta.crossover(close, nwe_lower) and use_nwe
nwe_sell_signal = ta.crossunder(close, nwe_upper) and use_nwe

// Определение кластеров
bullish_cluster = high_volume and delta > 0 and strong_delta and low < val_price and close > open
bearish_cluster = high_volume and delta < 0 and strong_delta and high > vah_price and close < open

// Реакция цены на кластер (для подтверждения)
bullish_reaction = bullish_cluster[1] and close > open
bearish_reaction = bearish_cluster[1] and close < open

// Условия для входа в позицию
// Интегрируем сигналы Nadaraya-Watson Envelope
long_condition = (poc_direction > 0 and bullish_cluster) or (poc_direction > 0 and nwe_buy_signal)
short_condition = (poc_direction < 0 and bearish_cluster) or (poc_direction < 0 and nwe_sell_signal)

// Ограничение количества сделок в день
var int trades_today = 0
if (dayofweek != dayofweek[1])
    trades_today := 0

can_trade = trades_today < max_trades_per_day

// Расчет размера позиции
// Ограничиваем размер позиции до 2% от депозита
position_size = math.min(strategy.equity * (risk_percent / 100) / (atr_value * stop_loss_atr), strategy.equity * 0.05) // лимит увеличен до 5%
position_size_dollars = position_size * close

// Расчет уровней стоп-лосса и тейк-профита
// Устанавливаем соотношение риск/прибыль минимум 2.5
long_stop = low - atr_value * stop_loss_atr
long_take = close + atr_value * take_profit_atr // take_profit_atr уже установлен на 5.0, что дает соотношение 2.5:1
short_stop = high + atr_value * stop_loss_atr
short_take = close - atr_value * take_profit_atr

// Трейлинг-стоп параметры
// Увеличиваем трейлинг-стоп для сохранения прибыли
trail_offset = atr_value * stop_loss_atr * trail_offset_mult // теперь настраивается через параметр
trail_step = atr_value * trailing_step

// Выполнение торговых операций
if (long_condition and can_trade)
    strategy.entry("Long", strategy.long, qty=position_size, comment="Long Entry")
    if (use_trailing_stop)
        strategy.exit("Exit Long", "Long", trail_points=trail_offset, trail_offset=trail_offset, comment="Trailing Stop")
    else
        strategy.exit("Exit Long", "Long", stop=long_stop, limit=long_take, comment="SL/TP")
    trades_today := trades_today + 1

if (short_condition and can_trade)
    strategy.entry("Short", strategy.short, qty=position_size, comment="Short Entry")
    if (use_trailing_stop)
        strategy.exit("Exit Short", "Short", trail_points=trail_offset, trail_offset=trail_offset, comment="Trailing Stop")
    else
        strategy.exit("Exit Short", "Short", stop=short_stop, limit=short_take, comment="SL/TP")
    trades_today := trades_today + 1

// Отображение Dynamic Levels Channel
plot(poc_price, color=color.blue, linewidth=2, title="POC")
plot(vah_price, color=color.red, linewidth=1, title="VAH")
plot(val_price, color=color.red, linewidth=1, title="VAL")
fill(plot(vah_price, display=display.none), plot(val_price, display=display.none), color=color.new(color.blue, 90), title="Value Area")

// Отображение Nadaraya-Watson Envelope
plot(use_nwe and not repaint ? nwe_upper : na, title="NWE Upper", color=color.teal, linewidth=1)
plot(use_nwe and not repaint ? nwe_lower : na, title="NWE Lower", color=color.red, linewidth=1)

// Отображение сигналов от NWE
plotshape(nwe_buy_signal, title="NWE Buy Signal", location=location.belowbar, color=color.teal, style=shape.triangleup, size=size.small)
plotshape(nwe_sell_signal, title="NWE Sell Signal", location=location.abovebar, color=color.red, style=shape.triangledown, size=size.small)

// Отображение кластеров
plotshape(bullish_cluster, title="Бычий кластер", location=location.belowbar, color=color.green, style=shape.square, size=size.normal)
plotshape(bearish_cluster, title="Медвежий кластер", location=location.abovebar, color=color.red, style=shape.square, size=size.normal)

// Отображение сигналов
plotshape(long_condition and can_trade, title="Сигнал на покупку", location=location.belowbar, color=color.green, style=shape.triangleup, size=size.normal)
plotshape(short_condition and can_trade, title="Сигнал на продажу", location=location.abovebar, color=color.red, style=shape.triangledown, size=size.normal)

// Отображение направления тренда
bgcolor(poc_direction > 0 ? color.new(color.green, 95) : poc_direction < 0 ? color.new(color.red, 95) : na)

// Отображение стоп-лоссов и тейк-профитов для открытых позиций
var line long_sl_line = na
var line long_tp_line = na
var line short_sl_line = na
var line short_tp_line = na

if (strategy.position_size > 0)
    if (not na(long_sl_line))
        line.delete(long_sl_line)
    if (not na(long_tp_line))
        line.delete(long_tp_line)
    
    long_sl_line := line.new(x1=bar_index, y1=long_stop, x2=bar_index + 20, y2=long_stop, color=color.red, width=1, style=line.style_dashed)
    long_tp_line := line.new(x1=bar_index, y1=long_take, x2=bar_index + 20, y2=long_take, color=color.green, width=1, style=line.style_dashed)
    
    label.new(x=bar_index, y=long_stop, text="SL", color=color.red, style=label.style_label_up, textcolor=color.white, size=size.small)
    label.new(x=bar_index, y=long_take, text="TP", color=color.green, style=label.style_label_down, textcolor=color.white, size=size.small)

if (strategy.position_size < 0)
    if (not na(short_sl_line))
        line.delete(short_sl_line)
    if (not na(short_tp_line))
        line.delete(short_tp_line)
    
    short_sl_line := line.new(x1=bar_index, y1=short_stop, x2=bar_index + 20, y2=short_stop, color=color.red, width=1, style=line.style_dashed)
    short_tp_line := line.new(x1=bar_index, y1=short_take, x2=bar_index + 20, y2=short_take, color=color.green, width=1, style=line.style_dashed)
    
    label.new(x=bar_index, y=short_stop, text="SL", color=color.red, style=label.style_label_down, textcolor=color.white, size=size.small)
    label.new(x=bar_index, y=short_take, text="TP", color=color.green, style=label.style_label_up, textcolor=color.white, size=size.small)

// Добавляем отладочную информацию
var label debug_info = label.new(bar_index, high, "", color=color.blue, style=label.style_label_down, size=size.small)
if (barstate.islast)
    label.set_xy(debug_info, bar_index, high)
    label.set_text(debug_info, "POC Dir: " + str.tostring(poc_direction) + 
                              "\nBull Cluster: " + str.tostring(bullish_cluster) + 
                              "\nBear Cluster: " + str.tostring(bearish_cluster) + 
                              "\nBull React: " + str.tostring(bullish_reaction) + 
                              "\nBear React: " + str.tostring(bearish_reaction) + 
                              "\nNWE Buy: " + str.tostring(nwe_buy_signal) + 
                              "\nNWE Sell: " + str.tostring(nwe_sell_signal) + 
                              "\nLong Cond: " + str.tostring(long_condition) + 
                              "\nShort Cond: " + str.tostring(short_condition) + 
                              "\nCan Trade: " + str.tostring(can_trade))

// Отображение статистики
var table info = table.new(position.top_right, 5, 12, color.black, color.white, 1, color.gray, 1)
if (barstate.islastconfirmedhistory)
    table.cell(info, 0, 0, "Стратегия", text_color=color.white, bgcolor=color.blue)
    table.cell(info, 1, 0, "Dynamic Levels", text_color=color.white, bgcolor=color.blue)
    table.cell(info, 0, 1, "Прибыль", text_color=color.white, bgcolor=color.blue)
    table.cell(info, 1, 1, str.tostring(strategy.netprofit), text_color=color.white)
    table.cell(info, 0, 2, "Просадка", text_color=color.white, bgcolor=color.blue)
    table.cell(info, 1, 2, str.tostring(strategy.max_drawdown), text_color=color.white)
    table.cell(info, 0, 3, "Сделок", text_color=color.white, bgcolor=color.blue)
    table.cell(info, 1, 3, str.tostring(strategy.closedtrades), text_color=color.white)
    table.cell(info, 0, 4, "Винрейт", text_color=color.white, bgcolor=color.blue)
    table.cell(info, 1, 4, str.tostring(strategy.wintrades / math.max(1, strategy.closedtrades) * 100) + "%", text_color=color.white)
    table.cell(info, 0, 5, "Сделок сегодня", text_color=color.white, bgcolor=color.blue)
    table.cell(info, 1, 5, str.tostring(trades_today), text_color=color.white)
    
    // Добавляем информацию о размере позиции
    table.cell(info, 0, 6, "Размер позиции", text_color=color.white, bgcolor=color.blue)
    table.cell(info, 1, 6, str.tostring(position_size, "#.####"), text_color=color.white)
    table.cell(info, 0, 7, "Размер в $", text_color=color.white, bgcolor=color.blue)
    table.cell(info, 1, 7, str.tostring(position_size_dollars, "#.##"), text_color=color.white)
    
    // Добавляем информацию о стоп-лоссах и тейк-профитах
    table.cell(info, 0, 8, "Long SL", text_color=color.white, bgcolor=color.blue)
    table.cell(info, 1, 8, str.tostring(long_stop, "#.####"), text_color=color.white)
    table.cell(info, 0, 9, "Long TP", text_color=color.white, bgcolor=color.blue)
    table.cell(info, 1, 9, str.tostring(long_take, "#.####"), text_color=color.white)
    table.cell(info, 0, 10, "Short SL", text_color=color.white, bgcolor=color.blue)
    table.cell(info, 1, 10, str.tostring(short_stop, "#.####"), text_color=color.white)
    table.cell(info, 0, 11, "Short TP", text_color=color.white, bgcolor=color.blue)
    table.cell(info, 1, 11, str.tostring(short_take, "#.####"), text_color=color.white)

// Ограничение риска
strategy.risk.max_intraday_filled_orders(max_trades_per_day)
